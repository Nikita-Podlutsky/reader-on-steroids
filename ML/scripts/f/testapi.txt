import requests
import json

API_BASE_URL = "http://127.0.0.1:8000"

def rank_documents(query: str, documents: list) -> dict | None:
    endpoint = f"{API_BASE_URL}/rank"
    payload = {"query": query, "documents": documents}
    
    print("="*50)
    print(f"Отправка запроса на: {endpoint}")
    print("Тело запроса (payload):\n", json.dumps(payload, indent=2, ensure_ascii=False))

    try:
        response = requests.post(endpoint, json=payload, timeout=60)
        response.raise_for_status()
        return response.json()
    except requests.exceptions.RequestException as e:
        print(f"\n❌ Ошибка при запросе к API: {e}")
        if e.response is not None: print("Детали ответа от сервера:", e.response.text)
        return None

if __name__ == "__main__":
    search_query = "What is Generative Adversarial Imitation Learning?"

    # --- ТЕПЕРЬ ИСПОЛЬЗУЕМ ТОЛЬКО 'content' ---
    # Я скопировал краткие выдержки из реальных статей
    candidate_docs = [
        {
          "content": "This paper introduces Generative Adversarial Imitation Learning (GAIL), a new method for model-free imitation learning which is able to learn policies from expert trajectories. GAIL is based on a generator-discriminator framework similar to Generative Adversarial Networks (GANs)."
        },
        {
          "content": "We propose a method for training deep generative models that we call Generative Adversarial Networks (GANs). In this approach, we simultaneously train two models: a generative model G that captures the data distribution, and a discriminative model D that estimates the probability that a sample came from the training data rather than G."
        },
        {
          "content": "The Transformer, the first sequence transduction model based entirely on attention, replaces the recurrent layers most commonly used in encoder-decoder architectures with multi-headed self-attention. For translation tasks, the Transformer can be trained significantly faster than architectures based on recurrent or convolutional layers."
        }
    ]

    api_response = rank_documents(search_query, candidate_docs)

    if api_response:
        print("\n" + "="*50)
        print("✅ Результаты ранжирования получены успешно!")
        
        scores = api_response.get("scores", [])
        results = []
        for doc, score in zip(candidate_docs, scores):
            doc_repr = f"'{doc.get('content', '')[:40]}...'"
            results.append({"document": doc_repr, "score": score})
            
        sorted_results = sorted(results, key=lambda x: x["score"], reverse=True)
        
        print("\n--- Отсортированный Ранжированный Список ---")
        for i, res in enumerate(sorted_results):
            print(f"{i+1}. Документ: {res['document']:<45} | Оценка (Score): {res['score']:.4f}")
        print("="*50)
